cmake_minimum_required(VERSION 3.21)
project(markdown-glass VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
# For clangd, clang-tidy and other tools that index the code.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(MDGLASS_SANITIZE "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(MDGLASS_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

option(MDGLASS_FUZZ "Build libFuzzer targets (requires clang)" OFF)
if(MDGLASS_FUZZ)
    add_compile_options(-fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()

include(CTest)

# Warnings for our own code (not the vendored parser). Left out on purpose:
# -Wconversion (Qt mixes int and qreal so freely that it buries real problems)
# and -Wnull-dereference (false alarms from inlined Qt container code).
set(MDGLASS_WARNINGS -Wall -Wextra -Wshadow -Wnon-virtual-dtor -Woverloaded-virtual
    -Wcast-align -Wimplicit-fallthrough -Wmissing-declarations)
# CI turns warnings into errors for our code only; the vendored parser keeps
# its own flags.
option(MDGLASS_WERROR "Treat compiler warnings in Markdown Glass's own code as errors" OFF)
if(MDGLASS_WERROR)
    list(APPEND MDGLASS_WARNINGS -Werror)
endif()

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets PrintSupport Network)
if(BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
endif()
find_package(KF6SyntaxHighlighting REQUIRED)

add_subdirectory(third_party)
add_subdirectory(src)

if(MDGLASS_FUZZ)
    add_subdirectory(fuzz)
endif()

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

include(GNUInstallDirs)
install(FILES data/io.github.markdown_glass.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
# The AppStream release list is generated from CHANGELOG.md.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS CHANGELOG.md)
execute_process(COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/scripts/package/changelog.sh metainfo
                OUTPUT_VARIABLE MDGLASS_RELEASES OUTPUT_STRIP_TRAILING_WHITESPACE
                COMMAND_ERROR_IS_FATAL ANY)
set(RELEASES "${MDGLASS_RELEASES}")
configure_file(data/io.github.markdown_glass.metainfo.xml.in io.github.markdown_glass.metainfo.xml @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/io.github.markdown_glass.metainfo.xml
        DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)

# Every package carries the licence of the program and of the parser compiled
# into it (its licence file is COPYING in the vendored tree; a plain name
# here says what it belongs to), plus the README.
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES third_party/cmark-gfm/COPYING DESTINATION ${CMAKE_INSTALL_DOCDIR} RENAME LICENSE.cmark-gfm)

# Debian packages also carry the changelog, generated by
# scripts/package/build-deb.sh and passed in here, and the DEP-5 copyright
# file Debian policy asks for (packaging/copyright, as /usr/share/doc/<name>/copyright).
set(MDGLASS_DEBIAN_CHANGELOG "" CACHE FILEPATH "changelog.Debian.gz to install with the .deb")
if(MDGLASS_DEBIAN_CHANGELOG)
    install(FILES ${MDGLASS_DEBIAN_CHANGELOG} DESTINATION ${CMAKE_INSTALL_DOCDIR})
endif()
set(MDGLASS_DEBIAN_COPYRIGHT "" CACHE FILEPATH "DEP-5 copyright file to install with the .deb")
if(MDGLASS_DEBIAN_COPYRIGHT)
    install(FILES ${MDGLASS_DEBIAN_COPYRIGHT} DESTINATION ${CMAKE_INSTALL_DOCDIR} RENAME copyright)
endif()
install(FILES data/io.github.markdown_glass.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)

# Debian and Ubuntu packages are made with CPack (see scripts/package/build-deb.sh).
# Library dependencies are worked out by dpkg-shlibdeps; the Qt plugins that
# are loaded at run time are added by hand.
set(CPACK_PACKAGE_NAME markdown-glass)
set(CPACK_PACKAGE_VENDOR "FingerlessGloves")
set(CPACK_PACKAGE_CONTACT "FingerlessGloves <me@fingerlessgloves.me>")
# The summary and description come from data/description.txt, shared with the
# RPM spec: summary on the first line, description after a blank line.
file(READ data/description.txt MDGLASS_DESCRIPTION_FILE)
string(REGEX MATCH "^[^\n]*" MDGLASS_SUMMARY "${MDGLASS_DESCRIPTION_FILE}")
string(REGEX REPLACE "^[^\n]*\n\n" "" MDGLASS_DESCRIPTION "${MDGLASS_DESCRIPTION_FILE}")
string(STRIP "${MDGLASS_DESCRIPTION}" MDGLASS_DESCRIPTION)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "${MDGLASS_SUMMARY}")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/FingerlessGlov3s/markdown-glass")
set(CPACK_STRIP_FILES ON)
set(CPACK_DEBIAN_PACKAGE_SECTION text)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-svg-plugins, qt6-image-formats-plugins, hicolor-icon-theme")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
# CPack adds the summary line and Debian's indentation itself.
set(CPACK_DEBIAN_PACKAGE_DESCRIPTION "${MDGLASS_DESCRIPTION}")
include(CPack)
